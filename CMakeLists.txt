cmake_minimum_required(VERSION 3.20)
project(tsn_c_stack VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(TSN_TESTS "Build the host tests" ON)
option(TSN_FREESTANDING "Build the RV32 minimal port and smoke test" OFF)
option(TSN_COVERAGE "Instrument release core tests with gcov" OFF)
option(TSN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Werror)
if(TSN_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(tsn STATIC src/adp.c src/acmp.c src/maap.c)
target_include_directories(tsn PUBLIC include)
if(TSN_FREESTANDING)
  if(TSN_TESTS OR TSN_COVERAGE OR TSN_SANITIZERS)
    message(FATAL_ERROR "The freestanding target needs hosted instrumentation disabled")
  endif()
  enable_language(ASM)
  target_include_directories(tsn PUBLIC examples/rv32/include)
  target_compile_definitions(tsn PRIVATE CTRL_REENTRY_ASSERT)
  add_executable(rv32_smoke examples/rv32/smoke.c examples/rv32/runtime.c examples/rv32/start.S)
  target_compile_definitions(rv32_smoke PRIVATE CTRL_REENTRY_ASSERT)
  target_link_libraries(rv32_smoke PRIVATE -Wl,--whole-archive tsn -Wl,--no-whole-archive gcc)
  target_link_options(rv32_smoke PRIVATE -nostdlib -nostartfiles
    "-T${CMAKE_CURRENT_SOURCE_DIR}/examples/rv32/link.ld" -Wl,--no-undefined -Wl,--fatal-warnings
    "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/rv32_smoke.map")
  set_target_properties(rv32_smoke PROPERTIES SUFFIX ".elf"
    LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/examples/rv32/link.ld")
endif()
if(TSN_TESTS)
  enable_language(CXX)
  enable_testing()
  find_package(GTest REQUIRED)
  add_library(tsn_test STATIC src/adp.c src/acmp.c src/maap.c)
  target_include_directories(tsn_test PUBLIC include)
  target_compile_definitions(tsn_test PRIVATE NDEBUG CTRL_REENTRY_ASSERT)
  if(TSN_COVERAGE)
    target_compile_options(tsn_test PRIVATE -O0 -g --coverage)
    target_link_options(tsn_test INTERFACE --coverage)
  endif()
  function(core_test name source library)
    add_executable(${name} ${source} tests/main.cpp)
    target_link_libraries(${name} PRIVATE ${library} GTest::gmock GTest::gtest)
    add_test(NAME ${name} COMMAND ${name})
  endfunction()
  core_test(adp_tests tests/test_adp.cpp tsn_test)
  core_test(acmp_tests tests/test_acmp.cpp tsn_test)
  core_test(maap_tests tests/test_maap.cpp tsn_test)
  add_library(example_port STATIC examples/adp_port.c)
  target_include_directories(example_port PUBLIC examples include)
  target_link_libraries(example_port PUBLIC tsn_test)
  core_test(port_tests tests/test_port.cpp example_port)
  core_test(adp_release tests/test_adp_reentry.cpp tsn_test)
  target_compile_definitions(adp_release PRIVATE ADP_TEST_RELEASE)
  add_library(tsn_debug STATIC src/adp.c src/maap.c)
  target_include_directories(tsn_debug PUBLIC include)
  target_compile_options(tsn_debug PRIVATE -UNDEBUG)
  core_test(adp_debug tests/test_adp_reentry.cpp tsn_debug)
  core_test(maap_debug tests/test_maap_debug.cpp tsn_debug)
endif()
